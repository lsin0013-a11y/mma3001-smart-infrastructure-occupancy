smart_infra.ventilation

Ventilation requirement and indoor CO2 for a mechanically ventilated room.

The required outdoor-air supply is the people present times a per-person rate (AS 1668.2 gives 10 L/s per person for office-type use, the assumption here).

To judge a ventilation strategy, the room's CO2 concentration is simulated with a well-mixed single-zone mass balance:

V dC/dt = G N(t) + Q(t) (C_out - C)

with room volume V (m3), people N, CO2 generation G per person (m3/s), outdoor-air supply Q (m3/s) and outdoor concentration C_out. Within a time step N and Q are constant, so the equation is linear with constant coefficients and has the exact solution

C(t + dt) = C_ss + (C(t) - C_ss) exp(-Q dt / V),   C_ss = C_out + G N / Q

(or C(t) + G N dt / V when Q = 0). simulate_co2() uses this by default. The explicit Euler method is also provided for comparison: it is only stable while Q dt / V < 2, and only accurate when Q dt / V is much smaller than 1.

 60def required_airflow(people: np.ndarray, rate_per_person: float = RATE_PER_PERSON) -> np.ndarray:
 61    """Outdoor air needed for the people present.
 62
 63    Parameters
 64    ----------
 65    people : array_like
 66        Number of people (may be fractional, e.g. an interval's mean headcount).
 67    rate_per_person : float, default 10.0
 68        Outdoor air per person, L/s.
 69
 70    Returns
 71    -------
 72    numpy.ndarray
 73        Required outdoor-air supply, L/s.
 74
 75    Raises
 76    ------
 77    ValueError
 78        If any value of ``people`` is negative.
 79    """
 80    people = np.asarray(people, dtype=float)
 81    if (people < 0).any():
 82        raise ValueError("people must be non-negative")
 83    return rate_per_person * people

 86def design_occupancy(headcount: np.ndarray, quantile: float = DESIGN_QUANTILE) -> int:
 87    """Number of people a fixed ventilation design is sized for.
 88
 89    The ``quantile`` of the headcount over occupied time, rounded up to whole
 90    people. Sizing for the very highest reading would let one brief crowd (or
 91    sensor glitch) set the airflow for every hour of the year; the 99.9th
 92    percentile covers all but the rarest peaks.
 93
 94    Parameters
 95    ----------
 96    headcount : array_like
 97        Headcount per equal time step (e.g. 1-min means). Zeros (empty room)
 98        are ignored.
 99    quantile : float, default 0.999
100        Share of occupied time the design covers, between 0 and 1.
101
102    Returns
103    -------
104    int
105        Design occupancy, people.
106
107    Raises
108    ------
109    ValueError
110        If ``quantile`` is outside (0, 1], any headcount is negative, or the
111        room is never occupied.
112    """
113    headcount = np.asarray(headcount, dtype=float)
114    if not 0 < quantile <= 1:
115        raise ValueError("quantile must be in (0, 1]")
116    if (headcount < 0).any():
117        raise ValueError("headcount must be non-negative")
118    occupied = headcount[headcount > 0]
119    if occupied.size == 0:
120        raise ValueError("the room is never occupied")
121    return int(np.ceil(np.quantile(occupied, quantile)))

124def fallback_airflow(usual_headcount: np.ndarray, design_people: float, threshold: float,
125                     rate_per_person: float = RATE_PER_PERSON) -> np.ndarray:
126    """Outdoor air to supply while the headcount sensor is known to be dead.
127
128    Without a live reading the controller cannot see how many people are in
129    the room, so the fallback is sized to be safe rather than efficient: full
130    design air whenever the room is usually in use (its average headcount for
131    that time of the week exceeds ``threshold``), and none otherwise.
132
133    Parameters
134    ----------
135    usual_headcount : array_like
136        Average headcount for each step's time of the week, from past data.
137    design_people : float
138        Design occupancy, people (see :func:`design_occupancy`).
139    threshold : float
140        Average headcount above which the room counts as usually in use.
141    rate_per_person : float, default 10.0
142        Outdoor air per person, L/s.
143
144    Returns
145    -------
146    numpy.ndarray
147        Outdoor-air supply, L/s: ``rate_per_person * design_people`` or 0.
148
149    Raises
150    ------
151    ValueError
152        If ``usual_headcount``, ``design_people`` or ``threshold`` is negative.
153    """
154    usual_headcount = np.asarray(usual_headcount, dtype=float)
155    if (usual_headcount < 0).any() or design_people < 0 or threshold < 0:
156        raise ValueError("usual_headcount, design_people and threshold must be non-negative")
157    return np.where(usual_headcount > threshold, rate_per_person * design_people, 0.0)

160def steady_state_co2(rate_per_person: float = RATE_PER_PERSON, generation: float = CO2_GENERATION,
161                     outdoor: float = OUTDOOR_CO2) -> float:
162    """CO2 level a room settles at when supplied exactly ``rate_per_person``.
163
164    With Q = rate x N, the steady state C_out + G N / Q does not depend on N.
165    At 10 L/s per person it is 420 + 0.0052 / 10 x 1e6 = 940 ppm, just under
166    the 1000 ppm limit, which is why that rate is a common design value.
167    """
168    return outdoor + generation / rate_per_person * 1e6

171def simulate_co2(
172    people: np.ndarray,
173    airflow: np.ndarray,
174    dt: float,
175    volume: float = ROOM_AREA * ROOM_HEIGHT,
176    generation: float = CO2_GENERATION,
177    outdoor: float = OUTDOOR_CO2,
178    initial: float | None = None,
179    method: str = "exact",
180) -> np.ndarray:
181    """Simulate indoor CO2 for piecewise-constant occupancy and airflow.
182
183    Parameters
184    ----------
185    people : array_like
186        People present in each time step.
187    airflow : array_like
188        Outdoor-air supply in each time step, L/s (same length as ``people``).
189    dt : float
190        Time step, seconds.
191    volume : float, default 810.0
192        Room volume, m3.
193    generation : float, default 0.0052
194        CO2 generated per person, L/s.
195    outdoor : float, default 420.0
196        Outdoor CO2, ppm.
197    initial : float, optional
198        CO2 at the start, ppm. Defaults to ``outdoor``.
199    method : {"exact", "euler"}, default "exact"
200        ``"exact"`` uses the analytical solution within each step;
201        ``"euler"`` uses the explicit Euler method.
202
203    Returns
204    -------
205    numpy.ndarray
206        CO2 at the end of each step, ppm (same length as ``people``).
207
208    Raises
209    ------
210    ValueError
211        If the inputs differ in length, contain negative values, ``dt`` or
212        ``volume`` is not positive, or ``method`` is unknown.
213    """
214    people = np.asarray(people, dtype=float)
215    q = np.asarray(airflow, dtype=float) / 1000.0           # m3/s
216    if people.shape != q.shape:
217        raise ValueError("people and airflow must have the same length")
218    if (people < 0).any() or (q < 0).any():
219        raise ValueError("people and airflow must be non-negative")
220    if dt <= 0 or volume <= 0:
221        raise ValueError("dt and volume must be positive")
222    if method not in ("exact", "euler"):
223        raise ValueError(f"Unknown method '{method}'")
224
225    source = generation / 1000.0 * people * 1e6               # ppm * m3/s added by occupants
226    c = outdoor if initial is None else float(initial)
227    out = np.empty(len(people))
228    if method == "euler":
229        for i in range(len(people)):
230            c += dt * (source[i] + q[i] * (outdoor - c)) / volume
231            out[i] = c
232        return out
233
234    # Exact solution of the linear ODE over each step; decay factor precomputed.
235    decay = np.exp(-q * dt / volume)
236    with np.errstate(divide="ignore", invalid="ignore"):
237        steady = np.where(q > 0, outdoor + source / q, np.inf)
238    for i in range(len(people)):
239        c = steady[i] + (c - steady[i]) * decay[i] if q[i] > 0 else c + source[i] * dt / volume
240        out[i] = c
241    return out

244def assess(people: np.ndarray, airflow: np.ndarray, co2: np.ndarray, dt: float,
245           limit: float = CO2_LIMIT, mask: np.ndarray | None = None) -> dict[str, float]:
246    """Energy and air-quality measures of a ventilation strategy.
247
248    Parameters
249    ----------
250    people, airflow, co2 : array_like
251        People, outdoor-air supply (L/s) and simulated CO2 (ppm) per step.
252    dt : float
253        Time step, seconds.
254    limit : float, default 1000.0
255        CO2 threshold, ppm.
256    mask : array_like of bool, optional
257        Steps to include (e.g. excluding sensor outages). All if ``None``.
258
259    Returns
260    -------
261    dict
262        - ``air_m3``: total outdoor air supplied, m3 (proxy for fan and
263          conditioning energy);
264        - ``required_m3``: outdoor air needed at the per-person rate, m3;
265        - ``person_hours_over_limit``: person-hours spent above ``limit``;
266        - ``share_occupied_time_over_limit``: share of occupied time above it;
267        - ``peak_ppm``: highest CO2 while occupied;
268        - ``mean_occupied_ppm``: mean CO2 while occupied.
269    """
270    people, airflow, co2 = (np.asarray(a, dtype=float) for a in (people, airflow, co2))
271    m = np.ones(len(people), bool) if mask is None else np.asarray(mask, bool)
272    people, airflow, co2 = people[m], airflow[m], co2[m]
273    occupied = people > 0
274    over = co2 > limit
275    return {
276        "air_m3": float(airflow.sum() * dt / 1000.0),
277        "required_m3": float(required_airflow(people).sum() * dt / 1000.0),
278        "person_hours_over_limit": float((people * over).sum() * dt / 3600.0),
279        "share_occupied_time_over_limit": float((over & occupied).sum() / max(occupied.sum(), 1)),
280        "peak_ppm": float(co2[occupied].max()) if occupied.any() else float("nan"),
281        "mean_occupied_ppm": float(co2[occupied].mean()) if occupied.any() else float("nan"),
282    }
